# Feather Browser v0.1 + v0.2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a working GTK4 browser (v0.1) with WebKit rendering and tab management, then add tab hibernation, adblock, and URL privacy stripping (v0.2).

**Architecture:** Three layers: GTK4 shell (UI) → `PageRenderer` trait (bridge) → WebKitGTK (rendering). Shell logic is kept GTK-free for testability; GTK view components are thin wrappers. Features (hibernation, adblock, privacy) are side-car structs that hook into the tab lifecycle and network layer. v0.3 (custom Feather engine) is a separate plan.

**Tech Stack:** Rust 2021, gtk4 0.9, webkit2gtk 2.0 (optional feature), tokio 1.x, serde 1.x, serde_json 1.x, glib (via gtk4)

**Spec:** `docs/superpowers/specs/2026-10-20-feather-browser-design.md`

// SPDX-License-Identifier: AGPL-3.0-or-later
use serde::{Deserialize, Serialize};
use std::fs;
use std::path::Path;

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(default)]
pub struct Settings {
    pub adblock_enabled: bool,
    pub auto_hibernate_delay_secs: u32,
    pub homepage: String,
    pub zoom_default: f64,
    pub renderer: String,
}

impl Default for Settings {
    fn default() -> Self {
        Self {
            adblock_enabled: true,
            auto_hibernate_delay_secs: 30,
            homepage: "https://tnhc.dev".to_string(),
            zoom_default: 1.0,
            renderer: "webkit".to_string(),
        }
    }
}

impl Settings {
    pub fn load() -> Self {
        let path = config_path();
        Self::load_from(&path)
    }

    pub fn load_from(path: &str) -> Self {
        fs::read_to_string(path)
            .ok()
            .and_then(|s| serde_json::from_str(&s).ok())
            .unwrap_or_default()
    }

    pub fn save(&self) -> Result<(), Box<dyn std::error::Error>> {
        let path = config_path();
        self.save_to(&path)
    }

    pub fn save_to(&self, path: &str) -> Result<(), Box<dyn std::error::Error>> {
        if let Some(parent) = Path::new(path).parent() {
            fs::create_dir_all(parent)?;
        }
        let json = serde_json::to_string_pretty(self)?;
        fs::write(path, json)?;
        Ok(())
    }
}

fn config_path() -> String {
    let home = std::env::var("HOME").unwrap_or_else(|_| ".".to_string());
    format!("{}/.config/feather/config.json", home)
}
```

// SPDX-License-Identifier: AGPL-3.0-or-later

/// The central rendering abstraction. Both WebKitRenderer and FeatherRenderer
/// implement this. The shell holds Box<dyn PageRenderer> per tab and never
/// imports webkit2gtk directly.
pub trait PageRenderer {
    // Navigation
    fn load_url(&self, url: &str);
    fn go_back(&self);
    fn go_forward(&self);
    fn reload(&self);
    fn stop(&self);

    // State queries
    fn current_url(&self) -> String;
    fn title(&self) -> String;
    fn can_go_back(&self) -> bool;
    fn can_go_forward(&self) -> bool;

    // GTK widget — embed into the tab content area
    fn widget(&self) -> gtk4::Widget;

    fn set_zoom(&self, factor: f64);

    // Hibernation lifecycle — called by HibernationManager
    fn suspend(&self);   // drops DOM/JS state, stashes URL
    fn resume(&self);    // reloads from stashed URL

    // Nexus integration (v1: Nexus-origin pages only)
    fn run_script(&self, js: &str);

    // Signal connectors — call once after construction
    fn connect_url_changed(&self, f: Box<dyn Fn(String) + 'static>);
    fn connect_title_changed(&self, f: Box<dyn Fn(String) + 'static>);
    fn connect_load_finished(&self, f: Box<dyn Fn() + 'static>);
}

// Re-export concrete types so callers only import bridge::*
#[cfg(feature = "webkit-renderer")]
pub use webkit_renderer::WebKitRenderer;

#[cfg(feature = "feather-renderer")]
pub use feather_renderer::FeatherRenderer;

pub use mock_renderer::MockRenderer;

#[cfg(feature = "webkit-renderer")]
pub mod webkit_renderer;

// SPDX-License-Identifier: AGPL-3.0-or-later
use super::PageRenderer;
use gtk4::prelude::*;
use std::cell::RefCell;
use std::rc::Rc;
use webkit2gtk::prelude::*;
use webkit2gtk::{LoadEvent, PolicyDecisionType, WebView};

struct Inner {
    webview: WebView,
    suspended_url: Option<String>,
}

/// WebKit-backed PageRenderer. Wraps webkit2gtk::WebView with interior
/// mutability so all PageRenderer methods take &self.
#[derive(Clone)]
pub struct WebKitRenderer {
    inner: Rc<RefCell<Inner>>,
}

impl WebKitRenderer {
    pub fn new() -> Self {
        let webview = WebView::new();
        // Sensible security defaults
        if let Some(settings) = webkit2gtk::prelude::WebViewExt::settings(&webview) {
            settings.set_enable_javascript(true);
            settings.set_enable_webgl(true);
            settings.set_enable_media(true);
            settings.set_allow_modal_dialogs(false);
        }
        Self {
            inner: Rc::new(RefCell::new(Inner { webview, suspended_url: None })),
        }
    }
}

impl PageRenderer for WebKitRenderer {
    fn load_url(&self, url: &str) {
        self.inner.borrow().webview.load_uri(url);
    }

    fn go_back(&self) {
        self.inner.borrow().webview.go_back();
    }

    fn go_forward(&self) {
        self.inner.borrow().webview.go_forward();
    }

    fn reload(&self) {
        self.inner.borrow().webview.reload();
    }

    fn stop(&self) {
        self.inner.borrow().webview.stop_loading();
    }

    fn current_url(&self) -> String {
        self.inner.borrow().webview.uri()
            .map(|s| s.to_string())
            .unwrap_or_default()
    }

    fn title(&self) -> String {
        self.inner.borrow().webview.title()
            .map(|s| s.to_string())
            .unwrap_or_default()
    }

    fn can_go_back(&self) -> bool {
        self.inner.borrow().webview.can_go_back()
    }

    fn can_go_forward(&self) -> bool {
        self.inner.borrow().webview.can_go_forward()
    }

    fn widget(&self) -> gtk4::Widget {
        self.inner.borrow().webview.clone().upcast()
    }

    fn set_zoom(&self, factor: f64) {
        self.inner.borrow().webview.set_zoom_level(factor);
    }

    fn suspend(&self) {
        let mut inner = self.inner.borrow_mut();
        let url = inner.webview.uri().map(|s| s.to_string());
        inner.suspended_url = url;
        inner.webview.stop_loading();
        inner.webview.load_uri("about:blank");
    }

    fn resume(&self) {
        let inner = self.inner.borrow();
        if let Some(ref url) = inner.suspended_url {
            inner.webview.load_uri(url);
        }
    }

    fn run_script(&self, js: &str) {
        let inner = self.inner.borrow();
        inner.webview.run_javascript(js, gtk4::gio::Cancellable::NONE, |_| {});
    }

    fn connect_url_changed(&self, f: Box<dyn Fn(String) + 'static>) {
        let inner = self.inner.borrow();
        inner.webview.connect_uri_notify(move |view| {
            if let Some(uri) = view.uri() {
                f(uri.to_string());
            }
        });
    }

    fn connect_title_changed(&self, f: Box<dyn Fn(String) + 'static>) {
        let inner = self.inner.borrow();
        inner.webview.connect_title_notify(move |view| {
            if let Some(title) = view.title() {
                f(title.to_string());
            }
        });
    }

    fn connect_load_finished(&self, f: Box<dyn Fn() + 'static>) {
        let inner = self.inner.borrow();
        inner.webview.connect_load_changed(move |_view, event| {
            if event == LoadEvent::Finished {
                f();
            }
        });
    }
}
```

pub struct TabManager {
    tabs: Vec<Tab>,
    active_id: Option<TabId>,
    next_id: TabId,
}

impl TabManager {
    pub fn new() -> Self {
        Self { tabs: Vec::new(), active_id: None, next_id: 1 }
    }

    /// Adds a new tab, makes it active, returns its id.
    pub fn add_tab(&mut self, url: &str, renderer: Box<dyn PageRenderer>) -> TabId {
        let id = self.next_id;
        self.next_id += 1;
        self.tabs.push(Tab::new(id, url, renderer));
        self.active_id = Some(id);
        id
    }

    /// Switches active tab. Noop if id not found.
    pub fn switch_to(&mut self, id: TabId) {
        if self.tabs.iter().any(|t| t.id == id) {
            self.active_id = Some(id);
        }
    }

    /// Removes tab by id. Guards against removing the last tab.
    /// If active tab removed, switches to the nearest remaining tab.
    pub fn remove_tab(&mut self, id: TabId) {
        if self.tabs.len() <= 1 {
            return;
        }
        let pos = match self.tabs.iter().position(|t| t.id == id) {
            Some(p) => p,
            None => return,
        };
        self.tabs.remove(pos);
        if self.active_id == Some(id) {
            let new_pos = pos.saturating_sub(1).min(self.tabs.len() - 1);
            self.active_id = self.tabs.get(new_pos).map(|t| t.id);
        }
    }

    pub fn active_tab(&self) -> Option<&Tab> {
        let id = self.active_id?;
        self.tabs.iter().find(|t| t.id == id)
    }

    pub fn active_tab_mut(&mut self) -> Option<&mut Tab> {
        let id = self.active_id?;
        self.tabs.iter_mut().find(|t| t.id == id)
    }

    pub fn tabs(&self) -> &[Tab] {
        &self.tabs
    }

    pub fn get_tab_mut(&mut self, id: TabId) -> Option<&mut Tab> {
        self.tabs.iter_mut().find(|t| t.id == id)
    }
}

// SPDX-License-Identifier: AGPL-3.0-or-later

/// Pure URL-processing logic — no GTK dependency, fully unit-testable.
pub struct AddressBarLogic;

impl AddressBarLogic {
    /// Converts raw address bar input to a loadable URL.
    /// - Trims whitespace
    /// - Adds https:// if no scheme present and input looks like a domain
    /// - Falls back to DuckDuckGo search for everything else
    pub fn clean_url(input: &str) -> String {
        let trimmed = input.trim();
        if trimmed.starts_with("https://") || trimmed.starts_with("http://")
            || trimmed.starts_with("about:") || trimmed.starts_with("file://")
        {
            return trimmed.to_string();
        }
        // Looks like a domain if it contains a dot and no spaces
        if !trimmed.contains(' ') && trimmed.contains('.') {
            return format!("https://{}", trimmed);
        }
        // Treat as search query
        let encoded = trimmed.replace(' ', "+");
        format!("https://duckduckgo.com/?q={}", encoded)
    }
}

use crate::shell::tab_manager::{Tab, TabId, TabState};
use gtk4::prelude::*;
use gtk4::{Box as GBox, Button, Label, Orientation};

pub struct TabBar {
    widget: GBox,
}

impl TabBar {
    pub fn new() -> Self {
        let hbox = GBox::new(Orientation::Horizontal, 2);
        hbox.set_margin_start(4);
        Self { widget: hbox }
    }

    pub fn widget(&self) -> gtk4::Widget { self.widget.clone().upcast() }

    /// Redraws the entire tab strip from the current tabs slice.
    /// Call this after any TabManager mutation.
    pub fn render<F1, F2, F3>(&self, tabs: &[Tab], active_id: Option<TabId>,
        on_click: F1, on_close: F2, on_new: F3)
    where
        F1: Fn(TabId) + Clone + 'static,
        F2: Fn(TabId) + Clone + 'static,
        F3: Fn() + 'static,
    {
        // Clear existing children
        while let Some(child) = self.widget.first_child() {
            self.widget.remove(&child);
        }

        for tab in tabs {
            let tab_id = tab.id;
            let is_active = Some(tab_id) == active_id;
            let is_hibernated = tab.state == TabState::Hibernated;

            let tab_box = GBox::new(Orientation::Horizontal, 2);
            let title = if is_hibernated {
                format!("❄ {}", &tab.title.chars().take(20).collect::<String>())
            } else {
                tab.title.chars().take(20).collect::<String>()
            };
            let label = Label::new(Some(if title.is_empty() { "New Tab" } else { &title }));
            let close_btn = Button::from_icon_name("window-close-symbolic");
            close_btn.set_margin_start(4);

            let click_cb = on_click.clone();
            let tab_btn = Button::new();
            tab_btn.set_child(Some(&label));
            if is_active {
                tab_btn.add_css_class("suggested-action");
            }
            tab_btn.connect_clicked(move |_| click_cb(tab_id));

            let close_cb = on_close.clone();
            close_btn.connect_clicked(move |_| close_cb(tab_id));

            tab_box.append(&tab_btn);
            tab_box.append(&close_btn);
            self.widget.append(&tab_box);
        }

        // New tab button
        let new_btn = Button::from_icon_name("tab-new-symbolic");
        let new_cb = on_new;
        new_btn.connect_clicked(move |_| new_cb());
        self.widget.append(&new_btn);
    }
}

    }

    /// Call on every tab switch. Marks old tab as Idle, resumes new tab if Hibernated.
    pub fn on_tab_switched(&mut self, old_id: Option<TabId>, new_id: TabId, tabs: &mut Vec<Tab>) {
        // Mark old tab as Idle, start idle timer
        if let Some(old) = old_id {
            if let Some(tab) = tabs.iter_mut().find(|t| t.id == old) {
                if tab.state == TabState::Active {
                    tab.state = TabState::Idle;
                    self.idle_since.insert(old, Instant::now());
                }
            }
        }
        // Resume new tab if it was hibernated
        if let Some(tab) = tabs.iter_mut().find(|t| t.id == new_id) {
            if tab.state == TabState::Hibernated {
                tab.renderer.resume();
            }
            tab.state = TabState::Active;
            self.idle_since.remove(&new_id);
        }
    }

    /// Call periodically (or after a timer) to hibernate overdue idle tabs.
    pub fn check_and_hibernate(&mut self, tabs: &mut Vec<Tab>) {
        let now = Instant::now();
        let overdue: Vec<TabId> = self.idle_since
            .iter()
            .filter(|(_, &since)| {
                if let Some(simulated) = self.simulated_elapsed {
                    let _ = simulated; // treat all as overdue
                    true
                } else {
                    now.duration_since(since) >= self.delay
                }
            })
            .map(|(&id, _)| id)
            .collect();

        for id in overdue {
            if let Some(tab) = tabs.iter_mut().find(|t| t.id == id && t.state == TabState::Idle) {
                tab.renderer.suspend();
                tab.state = TabState::Hibernated;
                self.idle_since.remove(&id);
            }
        }
    }
}

// SPDX-License-Identifier: AGPL-3.0-or-later
use std::collections::HashSet;
use std::fs;

pub struct AdBlocker {
    blocked_domains: HashSet<String>,
}

impl AdBlocker {
    /// Loads a hosts-format blocklist. Returns empty blocker if file missing.
    pub fn load(path: &str) -> Self {
        let blocked_domains = fs::read_to_string(path)
            .unwrap_or_default()
            .lines()
            .filter_map(parse_hosts_line)
            .collect();
        Self { blocked_domains }
    }

    /// Returns true if the URL's domain (or any parent domain) is in the blocklist.
    pub fn is_blocked(&self, url: &str) -> bool {
        let domain = match extract_domain(url) {
            Some(d) => d,
            None => return false,
        };
        // Check domain and all parent domains (e.g. stats.doubleclick.net → doubleclick.net)
        let parts: Vec<&str> = domain.split('.').collect();
        for i in 0..parts.len().saturating_sub(1) {
            let candidate = parts[i..].join(".");
            if self.blocked_domains.contains(&candidate) {
                return true;
            }
        }
        false
    }

    /// Writes a minimal default blocklist to config_dir/blocklist.txt if absent.
    pub fn bundle_default_blocklist(config_dir: &str) {
        let path = format!("{}/blocklist.txt", config_dir);
        if std::path::Path::new(&path).exists() { return; }
        let _ = fs::create_dir_all(config_dir);
        let default = include_str!("../../assets/default_blocklist.txt");
        let _ = fs::write(&path, default);
    }
}

fn parse_hosts_line(line: &str) -> Option<String> {
    let line = line.trim();
    if line.starts_with('#') || line.is_empty() { return None; }
    // Hosts format: "0.0.0.0 domain.com" or just "domain.com"
    let parts: Vec<&str> = line.split_whitespace().collect();
    if parts.len() >= 2 {
        Some(parts[1].to_lowercase())
    } else if parts.len() == 1 {
        Some(parts[0].to_lowercase())
    } else {
        None
    }
}

fn extract_domain(url: &str) -> Option<String> {
    // Fast path: find "://" then take up to next "/"
    let after_scheme = url.split_once("://")?.1;
    let host = after_scheme.split('/').next()?;
    // Strip port if present
    let host = host.split(':').next()?;
    if host.is_empty() { return None; }
    Some(host.to_lowercase())
}

// SPDX-License-Identifier: AGPL-3.0-or-later

const TRACKING_PARAMS: &[&str] = &[
    "utm_source", "utm_medium", "utm_campaign", "utm_term", "utm_content",
    "utm_id", "utm_reader", "utm_name", "utm_publ", "utm_swu",
    "fbclid", "gclid", "msclkid", "dclid", "twclid",
    "mc_cid", "mc_eid",
    "ref", "_ga", "_gl",
    "igshid",
];

pub struct Privacy;

impl Privacy {
    /// Strips known tracking query parameters from a URL.
    /// Returns the URL unchanged if it cannot be parsed.
    pub fn strip_tracking_params(url: &str) -> String {
        // Split on fragment first so we can restore it
        let (url_no_frag, fragment) = match url.split_once('#') {
            Some((u, f)) => (u, Some(f)),
            None => (url, None),
        };

        let (base, query_str) = match url_no_frag.split_once('?') {
            Some((b, q)) => (b, Some(q)),
            None => {
                // No query string — return original (with fragment if any)
                return match fragment {
                    Some(f) => format!("{}#{}", url_no_frag, f),
                    None => url.to_string(),
                };
            }
        };

        let clean_params: Vec<&str> = query_str
            .unwrap_or("")
            .split('&')
            .filter(|param| {
                let key = param.split('=').next().unwrap_or("");
                !TRACKING_PARAMS.iter().any(|&t| key.eq_ignore_ascii_case(t))
            })
            .collect();

        let result = if clean_params.is_empty() {
            base.to_string()
        } else {
            format!("{}?{}", base, clean_params.join("&"))
        };

        match fragment {
            Some(f) => format!("{}#{}", result, f),
            None => result,
        }
    }
}

```rust
// SPDX-License-Identifier: AGPL-3.0-or-later
#[cfg(feature = "feather-renderer")]
mod inner {
    use fontdue::{Font, FontSettings};

    const FONT_BYTES: &[u8] = include_bytes!("../../assets/NotoSans-Regular.ttf");

    pub struct GlyphBitmap {
        pub pixels: Vec<u8>,
        pub width: usize,
        pub height: usize,
        pub bearing_x: i32,
        pub bearing_y: i32,
    }

    pub struct FontMetrics {
        font: Font,
        pub size: f32,
        pub ascent: f32,
        pub descent: f32,
        pub line_height: f32,
        pub space_advance: f32,
    }

    impl FontMetrics {
        pub fn load(size: f32) -> Self {
            let font = Font::from_bytes(FONT_BYTES, FontSettings::default())
                .expect("embedded NotoSans font bytes are valid");
            let lm = font.horizontal_line_metrics(size)
                .unwrap_or(fontdue::LineMetrics {
                    ascent: size * 0.8,
                    descent: -(size * 0.2),
                    line_gap: 0.0,
                    new_line_size: size,
                });
            let ascent = lm.ascent;
            let descent = (-lm.descent).max(0.0);
            let line_height = (ascent + descent) * 1.2;
            let space_advance = font.metrics(' ', size).advance_width;
            FontMetrics { font, size, ascent, descent, line_height, space_advance }
        }

        pub fn advance_width(&self, ch: char) -> f32 {
            self.font.metrics(ch, self.size).advance_width
        }

        pub fn measure_str(&self, s: &str) -> f32 {
            s.chars().map(|c| self.advance_width(c)).sum()
        }

        pub fn rasterize(&self, ch: char) -> GlyphBitmap {
            let (metrics, pixels) = self.font.rasterize(ch, self.size);
            GlyphBitmap {
                pixels,
                width: metrics.width,
                height: metrics.height,
                bearing_x: metrics.xmin,
                // bearing_y: distance from baseline to top of glyph (positive = above baseline)
                bearing_y: metrics.ymin + metrics.height as i32,
            }
        }
    }
}

#[cfg(feature = "feather-renderer")]
pub use inner::{FontMetrics, GlyphBitmap};
```

```rust
    let name = self.parse_identifier();
    if name.is_empty() { self.skip_to(';'); self.consume_if(';'); return vec![]; }
```

```rust
fn expand_shorthand(prop: &str, raw: &str) -> Vec<(String, Value)> {
    match prop {
        "margin" | "padding" => {
            let values: Vec<Value> = raw.split_whitespace()
                .map(CssParser::parse_value)
                .collect();
            expand_box_model(prop, &values)
        }
        "border-width" => {
            let values: Vec<Value> = raw.split_whitespace()
                .map(CssParser::parse_value)
                .collect();
            expand_box_model("border", &values)
        }
        "font" => expand_font_shorthand(raw),
        _ => vec![(prop.to_string(), CssParser::parse_value(raw))],
    }
}

fn expand_box_model(prefix: &str, values: &[Value]) -> Vec<(String, Value)> {
    if values.is_empty() { return vec![]; }
    let (top, right, bottom, left) = match values.len() {
        1 => (values[0].clone(), values[0].clone(), values[0].clone(), values[0].clone()),
        2 => (values[0].clone(), values[1].clone(), values[0].clone(), values[1].clone()),
        3 => (values[0].clone(), values[1].clone(), values[2].clone(), values[1].clone()),
        _ => (values[0].clone(), values[1].clone(), values[2].clone(), values[3].clone()),
    };
    let suffix = if prefix == "border" { "-width" } else { "" };
    vec![
        (format!("{}-top{}", prefix, suffix),    top),
        (format!("{}-right{}", prefix, suffix),  right),
        (format!("{}-bottom{}", prefix, suffix), bottom),
        (format!("{}-left{}", prefix, suffix),   left),
    ]
}

fn expand_font_shorthand(raw: &str) -> Vec<(String, Value)> {
    let mut decls = Vec::new();
    let parts: Vec<&str> = raw.split_whitespace().collect();
    for part in &parts {
        if part.ends_with("px") || part.ends_with("em") || part.ends_with("rem") {
            let v = CssParser::parse_value(part);
            if matches!(v, Value::Length(_, _)) {
                decls.push(("font-size".to_string(), v));
                break;
            }
        }
    }
    for part in &parts {
        match *part {
            "bold" => { decls.push(("font-weight".to_string(), Value::Keyword("bold".to_string()))); break; }
            "normal" => { decls.push(("font-weight".to_string(), Value::Keyword("normal".to_string()))); break; }
            _ => {}
        }
    }
    decls
}
```

Add method inside `impl CssParser`:
```rust
fn parse_compound_selector(&mut self) -> Selector {
    let mut left = self.parse_simple_selector();
    loop {
        // Detect combinator: whitespace before a non-special char, or explicit '>'
        let had_whitespace = !self.eof() && self.input[self.pos].is_whitespace();
        self.skip_whitespace_and_comments();
        if self.eof() || self.current() == '{' || self.current() == ',' {
            break;
        }
        let combinator = if self.current() == '>' {
            self.consume(); // '>'
            self.skip_whitespace_and_comments();
            Combinator::Child
        } else if had_whitespace
            && (self.current().is_alphanumeric()
                || matches!(self.current(), '#' | '.' | '*'))
        {
            Combinator::Descendant
        } else {
            break;
        };
        let right = self.parse_simple_selector();
        left = Selector {
            tag_name: right.tag_name,
            id: right.id,
            classes: right.classes,
            ancestor: Some((combinator, Box::new(left))),
        };
    }
    left
}
```

```rust
    }
    let properties = match &root.node_type {
        NodeType::Element(data) => specified_values(data, stylesheet, ancestors),
        _ => PropertyMap::new(),
    };
    let children = root.children.iter().map(|c| {
        let mut child_ancestors = ancestors.to_vec();
        if let NodeType::Element(data) = &root.node_type {
            child_ancestors.push(data.clone());
        }
        style_tree_depth(c, stylesheet, depth + 1, &child_ancestors)
    }).collect();
    StyledNode { node: root.clone(), properties, children }
}

fn specified_values(
    elem: &ElementData,
    stylesheet: &Stylesheet,
    ancestors: &[ElementData],
) -> PropertyMap {
    let mut rules: Vec<(&Rule, (usize, usize, usize))> = stylesheet
        .rules
        .iter()
        .filter_map(|rule| {
            rule.selectors
                .iter()
                .filter(|sel| selector_matches(elem, sel, ancestors))
                .map(|sel| sel.specificity())
                .max()
                .map(|spec| (rule, spec))
        })
        .collect();
    rules.sort_by_key(|(_, spec)| *spec);
    let mut map = PropertyMap::new();
    for (rule, _) in rules {
        for (name, value) in &rule.declarations {
            map.insert(name.clone(), value.clone());
        }
    }
    map
}

fn selector_matches(elem: &ElementData, sel: &Selector, ancestors: &[ElementData]) -> bool {
    if let Some(ref tag) = sel.tag_name {
        if elem.tag_name != *tag { return false; }
    }
    if let Some(ref id) = sel.id {
        if elem.attributes.get("id").map(|s| s.as_str()) != Some(id.as_str()) { return false; }
    }
    let class_attr = elem.attributes.get("class").map(|s| s.as_str()).unwrap_or("");
    let elem_classes: Vec<&str> = class_attr.split_whitespace().collect();
    for cls in &sel.classes {
        if !elem_classes.contains(&cls.as_str()) { return false; }
    }
    if let Some((combinator, ancestor_sel)) = &sel.ancestor {
        match combinator {
            Combinator::Child => {
                match ancestors.last() {
                    Some(parent) => {
                        if !selector_matches(parent, ancestor_sel, &ancestors[..ancestors.len() - 1]) {
                            return false;
                        }
                    }
                    None => return false,
                }
            }
            Combinator::Descendant => {
                let found = ancestors.iter().enumerate().any(|(i, anc)| {
                    selector_matches(anc, ancestor_sel, &ancestors[..i])
                });
                if !found { return false; }
            }
        }
    }
    true
}
```

```rust
#[cfg(feature = "feather-renderer")]
impl InlineFormattingContext {
    pub fn layout(&self, items: &[InlineText]) -> (Vec<LineBox>, f32) {
        use crate::engine::font::FontMetrics;

        if items.is_empty() { return (vec![], 0.0); }

        let font = FontMetrics::load(self.font_size);
        let line_height = font.line_height;
        let space_advance = font.space_advance;

        let mut lines: Vec<LineBox> = Vec::new();
        let mut current_items: Vec<PlacedInlineItem> = Vec::new();
        let mut cursor_x: f32 = 0.0;
        let mut line_y: f32 = 0.0;

        let flush = |current_items: &mut Vec<PlacedInlineItem>,
                     lines: &mut Vec<LineBox>,
                     line_y: &mut f32,
                     cursor_x: &mut f32,
                     line_height: f32,
                     baseline: f32| {
            if !current_items.is_empty() {
                lines.push(LineBox {
                    items: std::mem::take(current_items),
                    y: *line_y,
                    height: line_height,
                    baseline,
                });
                *line_y += line_height;
                *cursor_x = 0.0;
            }
        };

        let baseline = font.ascent;

        for item in items {
            let words: Vec<&str> = item.text.split_whitespace().collect();
            if words.is_empty() { continue; }
            for (i, word) in words.iter().enumerate() {
                let w = font.measure_str(word);
                let gap = if i == 0 && cursor_x == 0.0 { 0.0 } else { space_advance };
                if cursor_x + gap + w > self.available_width && cursor_x > 0.0 {
                    flush(&mut current_items, &mut lines, &mut line_y, &mut cursor_x, line_height, baseline);
                }
                let actual_gap = if cursor_x == 0.0 { 0.0 } else { space_advance };
                let x = cursor_x + actual_gap;
                current_items.push(PlacedInlineItem {
                    text: word.to_string(),
                    x,
                    width: w,
                    font_size: item.font_size,
                    color: item.color,
                });
                cursor_x = x + w;
            }
        }
        flush(&mut current_items, &mut lines, &mut line_y, &mut cursor_x, line_height, baseline);

        let total_height = line_y;
        (lines, total_height)
    }
}
```

```rust
pub fn blit_glyph(canvas: &mut Canvas, glyph: &GlyphBitmap, x: i32, baseline_y: i32, color: CssColor) {
    for gy in 0..glyph.height {
        for gx in 0..glyph.width {
            let alpha = glyph.pixels[gy * glyph.width + gx];
            if alpha == 0 { continue; }
            let cx = x + glyph.bearing_x + gx as i32;
            let cy = baseline_y - glyph.bearing_y + gy as i32;
            if cx < 0 || cy < 0 { continue; }
            let alpha_f = alpha as f32 / 255.0;
            let blended = CssColor {
                r: (color.r as f32 * alpha_f + 255.0 * (1.0 - alpha_f)) as u8,
                g: (color.g as f32 * alpha_f + 255.0 * (1.0 - alpha_f)) as u8,
                b: (color.b as f32 * alpha_f + 255.0 * (1.0 - alpha_f)) as u8,
                a: 255,
            };
            canvas.set_pixel(cx as usize, cy as usize, blended);
        }
    }
```

```rust
// SPDX-License-Identifier: AGPL-3.0-or-later
#[cfg(feature = "feather-renderer")]
mod inner {
    use fontdue::{Font, FontSettings};
    use std::cell::RefCell;
    use std::collections::HashMap;
    use std::sync::{Arc, OnceLock};

    const FONT_BYTES: &[u8] = include_bytes!("../../assets/NotoSans-Regular.ttf");

    static PARSED_FONT: OnceLock<Arc<Font>> = OnceLock::new();

    fn shared_font() -> Arc<Font> {
        PARSED_FONT
            .get_or_init(|| {
                Arc::new(
                    Font::from_bytes(FONT_BYTES, FontSettings::default())
                        .expect("embedded NotoSans font bytes are valid"),
                )
            })
            .clone()
    }

    thread_local! {
        static METRICS_CACHE: RefCell<HashMap<u32, FontMetrics>> =
            RefCell::new(HashMap::new());
    }

    pub struct GlyphBitmap {
        pub pixels: Vec<u8>,
        pub width: usize,
        pub height: usize,
        pub bearing_x: i32,
        pub bearing_y: i32,
    }

    #[derive(Clone)]
    pub struct FontMetrics {
        pub size: f32,
        pub ascent: f32,
        pub descent: f32,
        pub line_height: f32,
        pub space_advance: f32,
        font: Arc<Font>,
    }

    impl FontMetrics {
        pub fn load(size: f32) -> Self {
            let font = shared_font();
            let lm = font
                .horizontal_line_metrics(size)
                .unwrap_or(fontdue::LineMetrics {
                    ascent: size * 0.8,
                    descent: -(size * 0.2),
                    line_gap: 0.0,
                    new_line_size: size,
                });
            let ascent = lm.ascent;
            let descent = (-lm.descent).max(0.0);
            let line_height = (ascent + descent) * 1.2;
            let space_advance = font.metrics(' ', size).advance_width;
            FontMetrics { font, size, ascent, descent, line_height, space_advance }
        }

        pub fn load_cached(size: f32) -> Self {
            let key = size.to_bits();
            METRICS_CACHE.with(|cache| {
                let mut map = cache.borrow_mut();
                if let Some(fm) = map.get(&key) {
                    return fm.clone();
                }
                let fm = FontMetrics::load(size);
                map.insert(key, fm.clone());
                fm
            })
        }

        pub fn advance_width(&self, ch: char) -> f32 {
            self.font.metrics(ch, self.size).advance_width
        }

        pub fn measure_str(&self, s: &str) -> f32 {
            s.chars().map(|c| self.advance_width(c)).sum()
        }

        pub fn rasterize(&self, ch: char) -> GlyphBitmap {
            let (metrics, pixels) = self.font.rasterize(ch, self.size);
            GlyphBitmap {
                pixels,
                width: metrics.width,
                height: metrics.height,
                bearing_x: metrics.xmin,
                bearing_y: metrics.ymin + metrics.height as i32,
            }
        }
    }
}

#[cfg(feature = "feather-renderer")]
pub use inner::{FontMetrics, GlyphBitmap};
```

## Task 3: O(D×S) CSS Descendant Matching

**Files:**
- Modify: `src/engine/style.rs`
- Test: `tests/engine_css_test.rs` (append 2 tests)

**Interfaces:**
- Consumes: nothing new
- Produces: `selector_matches` internal signature unchanged; behaviour identical for all valid CSS; performance improved from O(D^S) to O(D×S)

**Context:** The current `selector_matches` recurses into itself for every ancestor position when checking descendant combinators — `selector_matches(anc, ancestor_sel, &ancestors[..i])` inside `.enumerate().any(...)`. For a selector with S levels and DOM depth D, this is O(D^S). The fix: extract `simple_matches` (checks tag/id/class, no recursion) and rewrite `selector_matches` as an iterative left-to-right walk through the ancestor chain. Note: `ElementData.attributes` (not `.attrs`) is the correct field name.

```rust
fn simple_matches(elem: &ElementData, sel: &Selector) -> bool {
    if let Some(ref tag) = sel.tag_name {
        if elem.tag_name != *tag { return false; }
    }
    if let Some(ref id) = sel.id {
        if elem.attributes.get("id").map(|s| s.as_str()) != Some(id.as_str()) {
            return false;
        }
    }
    let class_attr = elem.attributes.get("class").map(|s| s.as_str()).unwrap_or("");
    let elem_classes: Vec<&str> = class_attr.split_whitespace().collect();
    for cls in &sel.classes {
        if !elem_classes.contains(&cls.as_str()) { return false; }
    }
    true
}

fn selector_matches(elem: &ElementData, sel: &Selector, ancestors: &[ElementData]) -> bool {
    if !simple_matches(elem, sel) { return false; }
    let mut current = sel;
    let mut anc_slice = ancestors;
    while let Some((combinator, parent_sel)) = &current.ancestor {
        match combinator {
            Combinator::Child => {
                match anc_slice.last() {
                    Some(parent) if simple_matches(parent, parent_sel) => {
                        anc_slice = &anc_slice[..anc_slice.len() - 1];
                    }
                    _ => return false,
                }
            }
            Combinator::Descendant => {
                match anc_slice.iter().rposition(|a| simple_matches(a, parent_sel)) {
                    Some(i) => anc_slice = &anc_slice[..i],
                    None => return false,
                }
            }
        }
        current = parent_sel;
    }
    true
}
```

The `simple_matches` function goes immediately before `selector_matches`. The old `selector_matches` body (lines 67–98 in the original) is deleted entirely.

**Context:** `style_tree_depth` currently does `let mut child_ancestors = ancestors.to_vec()` for every DOM node — O(depth) allocation per node, O(nodes × depth) total. The fix: change the function to take `ancestors: &mut Vec<ElementData>`, push the current element before recursing, pop after. A single Vec reused across siblings.         true
```rust
    } else {
        false
    };
    let mut children = Vec::new();
    for c in &root.children {
        children.push(style_tree_depth(c, stylesheet, depth + 1, ancestors));
    }
    if pushed { ancestors.pop(); }
    StyledNode { node: root.clone(), properties, children }
}
```

Key points:
- `specified_values(data, stylesheet, ancestors)` — `ancestors` is `&mut Vec<ElementData>`; Rust auto-derefs it to `&[ElementData]` as required by `specified_values`'s existing signature. No change to `specified_values`.
- The explicit `for` loop (not `.map().collect()`) avoids the closure reborrow problem with `&mut`.
- `pushed` tracks whether this node contributed an element to `ancestors`; pop only when something was pushed.
